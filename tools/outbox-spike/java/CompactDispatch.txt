    public void process(APDU apdu) {
        byte[] b = apdu.getBuffer();
        if (selectingApplet()) {
            Util.arrayFillNonAtomic(allState, (short) 0, (short) allState.length, (byte) 0);
            if (cardLocked[0] == 2) CompactState.recover(cardReceipts, proofStorage, sha, scratch, (short) 0);
            b[0] = 0x43;
            b[1] = 0x50;
            b[2] = 0x52;
            b[3] = 0x4F;
            b[4] = 0x54;
            b[5] = 2;
            apdu.setOutgoingAndSend((short) 0, (short) 6);
            return;
        }
        if (b[ISO7816.OFFSET_CLA] != (byte) 0xB0 || b[ISO7816.OFFSET_P1] != 0 || b[ISO7816.OFFSET_P2] != 0) ISOException.throwIt(ISO7816.SW_INCORRECT_P1P2);
        byte ins = b[ISO7816.OFFSET_INS];
        // Private read/sign/cancel commands use short case-2 APDUs with Le=0.
        if ((ins == (byte) 0x74 || ins == (byte) 0x76 || ins == (byte) 0x79 || ins == (byte) 0x7A || ins == (byte) 0x7B || ins == (byte) 0x7D) && b[4] != 0) ISOException.throwIt(ISO7816.SW_WRONG_LENGTH);
        if (ins == (byte) 0x7D) {
            Util.setShort(b, (short) 0, JCSystem.getAvailableMemory(JCSystem.MEMORY_TYPE_PERSISTENT));
            Util.setShort(b, (short) 2, JCSystem.getAvailableMemory(JCSystem.MEMORY_TYPE_TRANSIENT_RESET));
            Util.setShort(b, (short) 4, JCSystem.getAvailableMemory(JCSystem.MEMORY_TYPE_TRANSIENT_DESELECT));
            Util.setShort(b, (short) 6, JCSystem.getMaxCommitCapacity());
            Util.setShort(b, (short) 8, JCSystem.getUnusedCommitCapacity());
            Util.setShort(b, (short) 10, (short) (cardLocked[0] == 2 ? cardReceipts.length : 0));
            Util.setShort(b, (short) 12, (short) (dleq == null ? 0 : dleq.length));
            b[14] = cardLocked[0];
            b[15] = 5;
            apdu.setOutgoingAndSend((short) 0, (short) 16);
            return;
        }
        if (ins == (byte) 0x7A) {
            short size = toCompressed(b, cardPubKey.getW(b, (short) 0));
            apdu.setOutgoingAndSend((short) 0, size);
            return;
        }
        if (ins == (byte) 0x7E) {
            if (cardLocked[0] != 0) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
            short len = compactReceive(apdu);
            CompactState.validateSetup(b, ISO7816.OFFSET_CDATA, len);
            // One private setup attempt. An allocation failure cannot be retried.
            cardLocked[0] = 1;
            dleq = new byte[(short) 12288];
            cardReceipts = new byte[(short) 16384];
            CompactState.configure(cardReceipts, b, ISO7816.OFFSET_CDATA);
            __CONSTANTS__
            short keyLen = toCompressed(b, cardPubKey.getW(b, (short) 0));
            if (keyLen != 33) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
            Util.arrayCopyNonAtomic(b, (short) 0, cardReceipts, CompactState.KEY, (short) 33);
            cardLocked[0] = 2;
            b[0] = 1;
            apdu.setOutgoingAndSend((short) 0, (short) 1);
            return;
        }
        if (cardLocked[0] != 2) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
        if (ins == (byte) 0x71 || ins == (byte) 0x72 || ins == (byte) 0x73 || ins == (byte) 0x74 || ins == (byte) 0x77 || ins == (byte) 0x78 || ins == (byte) 0x7B) allState[2] = 0;
        short active = (short) (allState[0] & 255);
        if (active != 0 && ins != (byte) 0x73 && ins != (byte) 0x74 && ins != (byte) 0x76 && ins != (byte) 0x7B) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
        CompactState.recover(cardReceipts, proofStorage, sha, scratch, active);
        if (ins == (byte) 0x71) {
            short len = compactReceive(apdu), slot = CompactState.stageLoad(cardReceipts, proofStorage, dleq, b, ISO7816.OFFSET_CDATA, len, scratch);
            short base = (short) (slot * PROOF_SIZE), hex = (short) (slot * HEX_LEN);
            toHex(proofStorage, (short) (base + PROOF_NONCE_OFFSET), (short) 32);
            Util.arrayCopyNonAtomic(scratch, X_HEX, slotHex, hex, (short) 64);
            toHex(proofStorage, (short) (base + PROOF_C_OFFSET), (short) 33);
            Util.arrayCopyNonAtomic(scratch, X_HEX, slotHex, (short) (hex + 64), (short) 66);
            CompactState.commitLoad(cardReceipts, proofStorage, slot, b, (short) (ISO7816.OFFSET_CDATA + 1));
            b[0] = (byte) slot;
            apdu.setOutgoingAndSend((short) 0, (short) 1);
            return;
        }
        if (ins == (byte) 0x78) {
            CompactState.keyset(cardReceipts, b, ISO7816.OFFSET_CDATA, compactReceive(apdu));
            b[0] = cardReceipts[CompactState.KEYS];
            apdu.setOutgoingAndSend((short) 0, (short) 1);
            return;
        }
        if (ins == (byte) 0x72) {
            short index = CompactState.begin(cardReceipts, proofStorage, b, ISO7816.OFFSET_CDATA, compactReceive(apdu), scratch);
            allState[0] = (byte) (index + 1);
            allState[1] = 0;
            b[0] = (byte) index;
            apdu.setOutgoingAndSend((short) 0, (short) 1);
            return;
        }
        if (ins == (byte) 0x73) {
            if (active == 0) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
            short done = CompactState.append(cardReceipts, (short) (active - 1), (short) (allState[1] & 255), b, ISO7816.OFFSET_CDATA, compactReceive(apdu), scratch);
            allState[1] = (byte) done;
            b[0] = (byte) done;
            apdu.setOutgoingAndSend((short) 0, (short) 1);
            return;
        }
        if (ins == (byte) 0x74) {
            if (active == 0) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
            short index = (short) (active - 1);
            CompactState.inputs(cardReceipts, proofStorage, index, (byte) 1, scratch);
            CompactState.outputs(cardReceipts, index, (short) (allState[1] & 255), scratch);
            // No public path exposes staging signatures, including after a crypto failure.
            allState[0] = 0;
            short first = (short) (CompactState.get(cardReceipts, index, (short) 1) & 255);
            short tail = secretTail(b, (short) (first * PROOF_SIZE));
            shaAll.reset();
            short n = (short) (CompactState.get(cardReceipts, index, (short) 0) & 255);
            for (short i = 0; i < n; i++) secretInto(shaAll, (short) (CompactState.get(cardReceipts, index, (short) (1 + 5 * i)) & 255), b, tail);
            short out = CompactState.outputStart(cardReceipts, index), count = (short) (CompactState.get(cardReceipts, index, (short) (out - 1)) & 255), outputRef = CompactState.cursor(cardReceipts, index, out);
            for (short i = 0; i < count; i++) {
                short off = (short) (out + 38 * i);
                CompactState.readCursor(cardReceipts, index, off, outputRef, scratch, X_NUM, (short) 4);
                short digits = toDecimal(scratch, X_NUM);
                shaAll.update(scratch, (short) (X_DEC + 10 - digits), digits);
                CompactState.readCursor(cardReceipts, index, (short) (off + 5), CompactState.forward(cardReceipts, index, outputRef, (short) 5), b, (short) 0, (short) 33);
                toHex(b, (short) 0, (short) 33);
                shaAll.update(scratch, X_HEX, (short) 66);
                if ((short) (i + 1) < count) outputRef = CompactState.forward(cardReceipts, index, outputRef, (short) 38);
            }
            shaAll.doFinal(b, (short) 0, (short) 0, scratch, X_MSG);
            short sig = schnorrHW.sign(cardPrivKey, cardPubKey, scratch, X_MSG, b, (short) 0);
            if (sig != 64) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
            CompactState.write(cardReceipts, index, CompactState.signatureStart(cardReceipts, index), b, (short) 0, (short) 64);
            // SIM-CUT payment-staged
            CompactState.hash(cardReceipts, proofStorage, index, sha, scratch, false);
            CompactState.rememberHash(cardReceipts, index, scratch);
            CompactState.hash(cardReceipts, proofStorage, index, sha, scratch, true);
            sig = schnorrHW.sign(cardPrivKey, cardPubKey, scratch, (short) 0, b, (short) 0);
            if (sig != 64) ISOException.throwIt(ISO7816.SW_CONDITIONS_NOT_SATISFIED);
            CompactState.write(cardReceipts, index, (short) (CompactState.signatureStart(cardReceipts, index) + 64), b, (short) 0, (short) 64);
            // SIM-CUT descriptor-staged
            CompactState.publish(cardReceipts, index);
            CompactState.recover(cardReceipts, proofStorage, sha, scratch, (short) 0);
            short len = CompactState.scope(cardReceipts, index, b);
            apdu.setOutgoingAndSend((short) 0, len);
            return;
        }
        if (ins == (byte) 0x75) {
            if (compactReceive(apdu) != 4) ISOException.throwIt(ISO7816.SW_WRONG_LENGTH);
            short index = (short) (b[5] & 255), offset = Util.getShort(b, (short) 6), count = (short) (b[8] & 255);
            if ((short)(allState[2] & 255) != (short)(index + 1)) {
                CompactState.verifyCommitted(cardReceipts, proofStorage, index, sha, scratch);
                allState[2] = (byte)(index + 1);
            }
            short len = CompactState.export(cardReceipts, proofStorage, index, offset, count, b);
            apdu.setOutgoingAndSend((short) 0, len);
            return;
        }
        if (ins == (byte) 0x7C) {
            if (compactReceive(apdu) != 1) ISOException.throwIt(ISO7816.SW_WRONG_LENGTH);
            short index = (short) (b[5] & 255);
            if ((short)(allState[2] & 255) != (short)(index + 1)) {
                CompactState.verifyCommitted(cardReceipts, proofStorage, index, sha, scratch);
                allState[2] = (byte)(index + 1);
            }
            short len = CompactState.scope(cardReceipts, index, b);
            apdu.setOutgoingAndSend((short) 0, len);
            return;
        }
        if (ins == (byte) 0x76) {
            short len = CompactState.status(cardReceipts, proofStorage, b, active);
            apdu.setOutgoingAndSend((short) 0, len);
            return;
        }
        if (ins == (byte) 0x77) {
            CompactState.retire(cardReceipts, b, ISO7816.OFFSET_CDATA, compactReceive(apdu));
            CompactState.recover(cardReceipts, proofStorage, sha, scratch, (short) 0);
            b[0] = 1;
            apdu.setOutgoingAndSend((short) 0, (short) 1);
            return;
        }
        if (ins == (byte) 0x79) {
            for (short i = 0; i < 64; i++) b[i] = cardReceipts[(short) (CompactState.HEAD + i * CompactState.HS)];
            apdu.setOutgoingAndSend((short) 0, (short) 64);
            return;
        }
        if (ins == (byte) 0x7B) {
            allState[0] = 0;
            CompactState.recover(cardReceipts, proofStorage, sha, scratch, (short) 0);
            b[0] = 1;
            apdu.setOutgoingAndSend((short) 0, (short) 1);
            return;
        }
        ISOException.throwIt(ISO7816.SW_INS_NOT_SUPPORTED);
    }
    private short compactReceive(APDU apdu) {
        short got = apdu.setIncomingAndReceive(), expected = apdu.getIncomingLength();
        if (expected > 255) ISOException.throwIt(ISO7816.SW_WRONG_LENGTH);
        while (got < expected) {
            short n = apdu.receiveBytes((short) (ISO7816.OFFSET_CDATA + got));
            if (n <= 0) ISOException.throwIt(ISO7816.SW_WRONG_LENGTH);
            got += n;
        }
        return got;
    }
